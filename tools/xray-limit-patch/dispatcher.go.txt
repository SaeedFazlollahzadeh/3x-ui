// Added to the pinned Xray dispatcher by the local build.
package dispatcher

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"time"

	"github.com/xtls/xray-core/common"
	"github.com/xtls/xray-core/common/buf"
	"github.com/xtls/xray-core/common/session"
	"github.com/xtls/xray-core/transport"
	"golang.org/x/time/rate"
)

type panelLimit struct {
	upload, download *rate.Limiter
}

var panelLimits = map[string]map[string]panelLimit{}

func init() {
	if len(os.Args) == 2 && os.Args[1] == "--xui-speed-limit-support" {
		fmt.Println("xui-speed-limits-v2")
		os.Exit(0)
	}
	path := os.Getenv("XUI_XRAY_SPEED_LIMIT_FILE")
	if path == "" {
		return
	}
	data, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	var config map[string]map[string]struct {
		Upload   int `json:"upload"`
		Download int `json:"download"`
	}
	if err := json.Unmarshal(data, &config); err != nil {
		panic(err)
	}
	for tag, entries := range config {
		limits := make(map[string]panelLimit)
		for email, speeds := range entries {
			limits[email] = panelLimit{upload: newPanelLimiter(speeds.Upload), download: newPanelLimiter(speeds.Download)}
		}
		panelLimits[tag] = limits
	}
}

func newPanelLimiter(bytes int) *rate.Limiter {
	if bytes <= 0 {
		return nil
	}
	return rate.NewLimiter(rate.Limit(bytes), max(1, bytes/10))
}

func applyPanelSpeedLimits(ctx context.Context, link *transport.Link) {
	inbound := session.InboundFromContext(ctx)
	if inbound == nil {
		return
	}
	limits := panelLimits[inbound.Tag]
	inboundLimit := limits[""]
	var clientLimit panelLimit
	if inbound.User != nil {
		clientLimit = limits[strings.ToLower(inbound.User.Email)]
	}
	if inboundLimit.upload != nil || clientLimit.upload != nil {
		link.Reader = &panelLimitedReader{Reader: link.Reader, ctx: ctx, inbound: inboundLimit.upload, client: clientLimit.upload}
	}
	if inboundLimit.download != nil || clientLimit.download != nil {
		// Freedom's Linux splice bypasses the dispatcher link writer.
		inbound.CanSpliceCopy = 3
		link.Writer = &panelLimitedWriter{Writer: link.Writer, ctx: ctx, inbound: inboundLimit.download, client: clientLimit.download}
	}
}

func waitPanelBytes(ctx context.Context, bytes int, inbound, client *rate.Limiter) error {
	for bytes > 0 {
		chunk := bytes
		if inbound != nil {
			chunk = min(chunk, inbound.Burst())
		}
		if client != nil {
			chunk = min(chunk, client.Burst())
			if err := client.WaitN(ctx, chunk); err != nil {
				return err
			}
		}
		if inbound != nil {
			if err := inbound.WaitN(ctx, chunk); err != nil {
				return err
			}
		}
		bytes -= chunk
	}
	return nil
}

type panelLimitedReader struct {
	buf.Reader
	ctx             context.Context
	inbound, client *rate.Limiter
}

func (r *panelLimitedReader) throttle(mb buf.MultiBuffer, readErr error) (buf.MultiBuffer, error) {
	if err := waitPanelBytes(r.ctx, int(mb.Len()), r.inbound, r.client); err != nil {
		buf.ReleaseMulti(mb)
		return nil, err
	}
	return mb, readErr
}

func (r *panelLimitedReader) ReadMultiBuffer() (buf.MultiBuffer, error) {
	return r.throttle(r.Reader.ReadMultiBuffer())
}

func (r *panelLimitedReader) ReadMultiBufferTimeout(timeout time.Duration) (buf.MultiBuffer, error) {
	if reader, ok := r.Reader.(buf.TimeoutReader); ok {
		return r.throttle(reader.ReadMultiBufferTimeout(timeout))
	}
	return nil, buf.ErrNotTimeoutReader
}

func (r *panelLimitedReader) Interrupt() { _ = common.Interrupt(r.Reader) }

type panelLimitedWriter struct {
	buf.Writer
	ctx             context.Context
	inbound, client *rate.Limiter
}

func (w *panelLimitedWriter) WriteMultiBuffer(mb buf.MultiBuffer) error {
	if err := waitPanelBytes(w.ctx, int(mb.Len()), w.inbound, w.client); err != nil {
		buf.ReleaseMulti(mb)
		return err
	}
	return w.Writer.WriteMultiBuffer(mb)
}

func (w *panelLimitedWriter) Close() error { return common.Close(w.Writer) }
func (w *panelLimitedWriter) Interrupt()   { _ = common.Interrupt(w.Writer) }
